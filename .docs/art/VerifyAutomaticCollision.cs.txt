int checks=0,failed=0,near=0;var details=new System.Text.StringBuilder();
System.Func<Vector2[],Vector2,bool> inside=(path,point)=>{bool value=false;for(int i=0,j=path.Length-1;i<path.Length;j=i++){var a=path[i];var b=path[j];if((a.y>point.y)!=(b.y>point.y)&&point.x<(b.x-a.x)*(point.y-a.y)/(b.y-a.y)+a.x)value=!value;}return value;};
foreach(var map in UnityEngine.Object.FindObjectsByType<UnityEngine.Tilemaps.Tilemap>(FindObjectsSortMode.None).Where(m=>m.name=="Ground"||m.name=="Ceiling")){
var tc=map.GetComponent<UnityEngine.Tilemaps.TilemapCollider2D>();tc.ProcessTilemapChanges();var composite=map.GetComponent<CompositeCollider2D>();composite.GenerateGeometry();var paths=new System.Collections.Generic.List<Vector2[]>();for(int n=0;n<composite.pathCount;n++){var pts=new Vector2[composite.GetPathPointCount(n)];composite.GetPath(n,pts);for(int j=0;j<pts.Length;j++)pts[j]=composite.transform.TransformPoint(pts[j]);paths.Add(pts);}
foreach(var p in map.cellBounds.allPositionsWithin){var auto=map.GetTile<PlatformBorderTile>(p);if(auto==null)continue;var transform=map.transform.localToWorldMatrix*Matrix4x4.Translate(map.CellToLocalInterpolated((Vector3)p+map.tileAnchor))*map.GetTransformMatrix(p);
for(int y=0;y<8;y++)for(int x=0;x<8;x++){var local=new Vector2((x+.5f)/8-.5f,(y+.5f)/8-.5f);bool expected=PlatformBorderTile.Contains(map.GetSprite(p),local);Vector2 world=transform.MultiplyPoint3x4(local);bool actual=false;foreach(var path in paths)if(inside(path,world))actual=!actual;checks++;if(actual!=expected){
float distance=99999;foreach(var path in paths)for(int i=0,j=path.Length-1;i<path.Length;j=i++){var v=path[i]-path[j];float t=v.sqrMagnitude<.000001f?0:Mathf.Clamp01(Vector2.Dot(world-path[j],v)/v.sqrMagnitude);distance=Mathf.Min(distance,Vector2.Distance(world,path[j]+t*v));}
if(distance<=.02f)near++;else {failed++;if(failed<10)details.AppendLine(map.name+" "+p+" "+local+" distance="+distance);}
}}
}}
return new{checks=checks,failuresBeyondExtrusion=failed,withinExtrusion=near,details=details.ToString()};
